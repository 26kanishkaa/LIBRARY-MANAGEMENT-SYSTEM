// script.js

const API_URL = "http://localhost:5000/books";

function loadBooks(){
  fetch(API_URL)
  .then(res => res.json())
  .then(data => {

    const table = document.getElementById("bookTable");
    table.innerHTML = "";

    data.forEach(book => {

      table.innerHTML += `
      <tr>
        <td>${book.category}</td>
        <td>${book.isbn}</td>
        <td>${book.title}</td>
        <td>${book.author}</td>
        <td>${book.status}</td>
      </tr>
      `;
    });

  });
}

function openModal(){
  document.getElementById("borrowModal").style.display="block";
}

function closeModal(){
  document.getElementById("borrowModal").style.display="none";
}

function borrowBook(){

  const studentId = document.getElementById("studentId").value;
  const isbn = document.getElementById("isbn").value;

  fetch("http://localhost:5000/borrow",{
    method:"POST",
    headers:{
      "Content-Type":"application/json"
    },
    body:JSON.stringify({
      studentId,
      isbn
    })
  })
  .then(res => res.json())
  .then(data => {
    alert(data.message);
    closeModal();
  });

}

loadBooks();
